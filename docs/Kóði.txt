import machine
import utime
import urandom
from machine import Pin, PWM, I2C
from lcd_api import LcdApi
from i2c_lcd import I2cLcd

# HARDWARE SETUP
I2C_ADDR = 0x27
i2c = I2C(0, sda=Pin(21), scl=Pin(1), freq=400000)
lcd = I2cLcd(i2c, I2C_ADDR, 2, 16)

button = Pin(4, Pin.IN, Pin.PULL_UP)
led = Pin(13, Pin.OUT)
reed_switch = Pin(5, Pin.IN, Pin.PULL_UP)

last_click_time = 0
double_click_delay = 450 # Slightly longer for easier double-clicking

# ROBUST SOUND FUNCTION
def make_sound(freq, duration):
    # We create and destroy the PWM object each time to prevent S3 glitches
    snd = PWM(Pin(18))
    snd.freq(freq)
    snd.duty_u16(32768)
    utime.sleep_ms(duration)
    snd.duty_u16(0)
    snd.deinit() # This "unplugs" the buzzer pin to stop glitches

def winner_celebration(msg):
    lcd.clear()
    lcd.putstr("!!! WINNER !!!")
    # Multi-tone Winning Siren
    for _ in range(3):
        led.value(1)
        make_sound(1000, 150)
        led.value(0)
        make_sound(1500, 150)
        led.value(1)
        make_sound(2000, 150)
        led.value(0)
        utime.sleep_ms(50)
    
    lcd.clear()
    lcd.putstr(msg)
    utime.sleep(2)

# STARTUP SEQUENCE 
lcd.clear()
lcd.putstr("F1 ENGINE START")
for f in [500, 800, 1200]:
    led.value(1)
    make_sound(f, 100)
    led.value(0)
    utime.sleep_ms(50)

led.value(1) # Keep button LED on
lcd.clear()
lcd.putstr("PRESS TO ROLL")

# MAIN LOOP
while True:
    # 1. Reed Switch (Magnet) Logic
    if reed_switch.value() == 0:
        winner_celebration("MAGNET VICTORY!")
        lcd.clear()
        lcd.putstr("PRESS TO ROLL")

    # 2. Button Logic
    if button.value() == 0:
        current_time = utime.ticks_ms()
        
        # Check for Double Click
        if utime.ticks_diff(current_time, last_click_time) < double_click_delay:
            # It's a win!
            winner_celebration("BUTTON VICTORY!")
            lcd.clear()
            lcd.putstr("PRESS TO ROLL")
        else:
            # It's a regular roll
            make_sound(1200, 50) # Short click sound
            roll = (urandom.getrandbits(8) % 6) + 1
            lcd.clear()
            lcd.putstr("YOU ROLLED: " + str(roll))
            
        last_click_time = current_time
        
        # Wait for release to prevent "spamming"
        while button.value() == 0:
            utime.sleep_ms(20)
            
    utime.sleep_ms(10)
